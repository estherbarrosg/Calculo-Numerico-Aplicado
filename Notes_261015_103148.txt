import numpy as np

def resolver_conducao_placa(n):
# Inicialização da matriz com (n+2) x (n+2) posições, preenchida com 20°C
T = np.full((n + 2, n + 2), 20.0)

# Condição de contorno inferior: T(i, 1) = 100°C para i = 2, ..., n+1
# No índice 0-based do Python, a linha inferior corresponde a j = 1
T[1:-1, 1] = 100.0

tol = 1e-6
max_iter = 100000
iter_count = 0
erro = 1.0

# Atualização iterativa por Gauss-Seidel
while erro > tol and iter_count < max_iter:
erro = 0.0
for j in range(2, n + 2):
for i in range(2, n + 2):
antigo = T[i, j]
# Média dos quatro nós vizinhos
T[i, j] = (T[i-1, j] + T[i+1, j] + T[i, j-1] + T[i, j+1]) / 4.0
diferenca = abs(T[i, j] - antigo)
if diferenca > erro:
erro = diferenca
iter_count += 1

convergiu = erro < tol

# Conferência adicional: cálculo do resíduo máximo r_ij
max_res = 0.0
for j in range(2, n + 2):
for i in range(2, n + 2):
r = 4.0 * T[i, j] - (T[i-1, j] + T[i+1, j] + T[i, j-1] + T[i, j+1])
if abs(r) > max_res:
max_res = abs(r)

return T, iter_count, erro, convergiu, max_res

Execução para n = 5, 10 e 20
for n in [5, 10, 20]:
T, iters, erro, conv, max_res = resolver_conducao_placa(n)
print(f"=== Resultados para n = {n} ===")
print(f"Número de iterações: {iters}")
print(f"Erro final: {erro:.2e}")
print(f"Convergiu: {conv}")
print(f"Resíduo máximo (max |r_ij|): {max_res:.2e}")
if n == 5:
print(f"Temperatura do nó central T(4,4): {T[4, 4]:.4f} °C")
print("-" * 45)
